# provisioning a network updater. A later online scope requires a reviewed
# contract and its own production trust/endpoint acceptance.

  file(READ "${CMAKE_SOURCE_DIR}/release-scope.json" TURAN_SCOPE_JSON)
  string(JSON TURAN_UPDATE_MODE GET "${TURAN_SCOPE_JSON}" productionUpdates)
  if(TURAN_UPDATE_MODE STREQUAL "disabled_until_provisioned")
    message(FATAL_ERROR "Offline release scope forbids TURAN_UPDATE_TRUST_HEADER for all channels; use a reviewed online acceptance scope")
  endif()
endif()

  target_compile_definitions(turan_update PRIVATE TURAN_UPDATE_TRUST_HEADER="${TURAN_UPDATE_TRUST_HEADER}")

  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/docs/TOR_ARCHITECTURE.md" "$<TARGET_FILE_DIR:TuranBrowser>/docs/TOR_ARCHITECTURE.md"

  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/docs/FILTER_SUBSCRIPTIONS.md" "$<TARGET_FILE_DIR:TuranBrowser>/docs/FILTER_SUBSCRIPTIONS.md"
