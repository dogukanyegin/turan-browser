cmake_minimum_required(VERSION 3.22)
project(TuranUpdateNativeTests LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Native updater tests require Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(CMAKE_CONFIGURATION_TYPES Release CACHE STRING "Release" FORCE)
get_filename_component(TURAN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(turan_update STATIC
  "${TURAN_ROOT}/src/update/manifest.cpp" "${TURAN_ROOT}/src/update/files.cpp"
  "${TURAN_ROOT}/src/update/download.cpp" "${TURAN_ROOT}/src/update/transaction.cpp"
  "${TURAN_ROOT}/src/update/trust.cpp" "${TURAN_ROOT}/src/update/worker.cpp")
target_include_directories(turan_update PUBLIC "${TURAN_ROOT}/src")
target_compile_definitions(turan_update PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(turan_update PRIVATE /utf-8 /W4)
target_link_libraries(turan_update PUBLIC bcrypt winhttp)
add_executable(turan_update_tests "${TURAN_ROOT}/tests/update_tests.cpp")
target_link_libraries(turan_update_tests PRIVATE turan_update)
target_compile_options(turan_update_tests PRIVATE /utf-8 /W4)
add_executable(TuranUpdateTool WIN32 "${TURAN_ROOT}/src/update/tool.cpp")
target_link_libraries(TuranUpdateTool PRIVATE turan_update shell32 user32)
target_compile_options(TuranUpdateTool PRIVATE /utf-8 /W4)
add_library(turan_update_ui STATIC "${TURAN_ROOT}/src/ui/update_window.cpp")
target_include_directories(turan_update_ui PUBLIC "${TURAN_ROOT}/src")
target_link_libraries(turan_update_ui PRIVATE turan_update user32 gdi32)
target_compile_options(turan_update_ui PRIVATE /utf-8 /W4)
target_link_libraries(turan_update_tests PRIVATE turan_update_ui user32)
enable_testing()
add_test(NAME turan_update COMMAND turan_update_tests "${TURAN_ROOT}/artifacts")
